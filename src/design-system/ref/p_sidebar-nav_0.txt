import * as React from 'react'
import {
  ChevronDown,
  type LucideIcon,
} from 'lucide-react'
import { cn } from '@/lib/utils'
import { Tooltip } from './tooltip'

export interface NavItem {
  id: string
  label: string
  icon: LucideIcon
  href?: string
  badge?: number | string
  children?: { id: string; label: string; href?: string; badge?: number | string }[]
}

export function SidebarNavItem({
  item,
  active,
  collapsed,
  expanded,
  onToggle,
  onSelect,
}: {
  item: NavItem
  active?: boolean
  collapsed?: boolean
  expanded?: boolean
  onToggle?: () => void
  onSelect?: (id: string) => void
}) {
  const Icon = item.icon
  const hasChildren = !!item.children?.length

  const button = (
    <button
      type="button"
      onClick={() => {
        if (hasChildren && !collapsed) onToggle?.()
        else onSelect?.(item.id)
      }}
      className={cn(
        'group flex w-full items-center gap-3 rounded-md px-3 py-2 font-body-md font-medium transition-colors duration-150',
        'focus-visible:outline-none focus-visible:shadow-[var(--shadow-focus)]',
        collapsed && 'justify-center px-0',
        active
          ? 'bg-[var(--color-sidebar-active)] text-[var(--color-sidebar-active-fg)]'
          : 'text-[var(--color-sidebar-fg)] hover:bg-[var(--color-sidebar-hover)] hover:text-white',
      )}
