/** Risk score bands (5×5 matrix default: score = likelihood × impact) */
export const riskSeverityBands = {
  low: { min: 1, max: 4, label: 'Low', scoreRange: '1–4' },
  moderate: { min: 5, max: 9, label: 'Moderate', scoreRange: '5–9' },
  high: { min: 10, max: 16, label: 'High', scoreRange: '10–16' },
  critical: { min: 17, max: 25, label: 'Critical', scoreRange: '17–25' },
} as const

export type RiskSeverity = keyof typeof riskSeverityBands

export function getRiskSeverity(score: number): RiskSeverity {
  if (score >= 17) return 'critical'
  if (score >= 10) return 'high'
  if (score >= 5) return 'moderate'
  return 'low'
}

  },

const severityIcons: Record<RiskSeverity, React.ElementType> = {
  low: ShieldCheck,
  moderate: Shield,
  high: ShieldAlert,
  critical: AlertTriangle,
}

export interface RiskSeverityBadgeProps {
  severity?: RiskSeverity
  score?: number
  showRange?: boolean
  showIcon?: boolean
  size?: 'sm' | 'md'
  className?: string
}

export function RiskSeverityBadge({
  severity,
  score,
  showRange = false,
  showIcon = false,
  size = 'md',
  className,
}: RiskSeverityBadgeProps) {
  const resolved = severity ?? (score != null ? getRiskSeverity(score) : 'low')
  const band = riskSeverityBands[resolved]
  const styles = severityStyles[resolved]
  const Icon = severityIcons[resolved]

  return (
    <span
      className={cn(
        'inline-flex items-center gap-1 rounded-md border font-caption font-medium whitespace-nowrap',
        size === 'sm' ? 'px-1.5 py-0 text-[11px]' : 'px-2 py-0.5',
        styles.bg,
        styles.text,
        styles.border,
        className,
      )}
      title={`${band.label}${showRange ? ` · ${band.scoreRange}` : ''}`}
    >
      {showIcon && <Icon className="size-3 shrink-0" aria-hidden />}
      <span
        className={cn('size-1.5 rounded-full shrink-0', styles.dot)}
        aria-hidden
      />
      <span>
        {band.label}
        {showRange && (
          <span className="opacity-70"> · {band.scoreRange}</span>
        )}
      </span>
    </span>
  )
}

export interface RiskScoreProps {
  score: number
  maxScore?: number
  likelihood?: number
  impact?: number
  variant?: 'inherent' | 'residual' | 'default'
  size?: 'sm' | 'md' | 'lg'
  showSeverity?: boolean
  className?: string
}

export function RiskScore({
  score,
  maxScore = 25,
  likelihood,
  impact,
  variant = 'default',
  size = 'md',
  showSeverity = true,
  className,
}: RiskScoreProps) {
  const severity = getRiskSeverity(score)
  const band = riskSeverityBands[severity]
  const styles = severityStyles[severity]

  const sizeClasses = {
    sm: 'text-[12px] gap-1',
    md: 'text-[13px] gap-1.5',
    lg: 'text-sm gap-2',
  }

  const scoreSize = {
    sm: 'font-semibold tabular-nums',
    md: 'font-semibold tabular-nums text-sm',
    lg: 'font-semibold tabular-nums text-base',
  }

  return (
    <div
      className={cn(
        'inline-flex items-center flex-wrap',
        sizeClasses[size],
        className,
      )}
      role="group"
      aria-label={`${variant !== 'default' ? variant + ' ' : ''}risk score ${score} out of ${maxScore}, ${band.label}`}
    >
      {variant !== 'default' && (
        <span className="font-overline text-[var(--color-text-muted)] mr-0.5">
          {variant === 'inherent' ? 'Inherent' : 'Residual'}
        </span>
      )}
      <span className={cn(scoreSize[size], styles.text)}>
        {score}
        <span className="text-[var(--color-text-muted)] font-normal">
          {' '}
          / {maxScore}
        </span>
      </span>
      {showSeverity && (
        <RiskSeverityBadge severity={severity} size={size === 'lg' ? 'md' : 'sm'} />
      )}
      {(likelihood != null || impact != null) && (
        <span className="font-caption text-[var(--color-text-muted)]">
          {likelihood != null && `L${likelihood}`}
          {likelihood != null && impact != null && ' · '}
          {impact != null && `I${impact}`}
        </span>
      )}
    </div>
  )
}

import { cn } from '@/lib/utils'
import { statusMeta, type StatusTone } from '@/tokens'
import { Badge } from './badge'

const toneToBadge = {
  neutral: 'neutral',
  info: 'info',
  success: 'success',
  warning: 'warning',
  error: 'error',
  primary: 'primary',
} as const

const toneDot: Record<StatusTone, string> = {
  neutral: 'bg-[var(--color-text-muted)]',
  info: 'bg-[var(--color-info)]',
  success: 'bg-[var(--color-success)]',
  warning: 'bg-[var(--color-warning)]',
  error: 'bg-[var(--color-error)]',
  primary: 'bg-[var(--color-primary)]',
}

export function getStatusMeta(status: string) {
  return statusMeta[status] ?? { label: status, tone: 'neutral' as StatusTone }
}

export interface StatusBadgeProps {
  status: string
  label?: string
  className?: string
  size?: 'sm' | 'md'
  showDot?: boolean
}

export function StatusBadge({
  status,
  label,
  className,
  size = 'md',
  showDot = true,
}: StatusBadgeProps) {
  const meta = getStatusMeta(status)
  const display = label ?? meta.label
  return (
    <Badge
      variant={toneToBadge[meta.tone]}
      size={size}
      className={className}
      title={display}
    >
      {showDot && (
        <span
          className={cn('size-1.5 rounded-full shrink-0', toneDot[meta.tone])}
          aria-hidden
        />
      )}
      <span>{display}</span>
    </Badge>
  )
}

export interface StatusDotProps {
  status: string
  className?: string
  withLabel?: boolean
}

export function StatusDot({ status, className, withLabel = true }: StatusDotProps) {
  const meta = getStatusMeta(status)
  return (
    <span className={cn('inline-flex items-center gap-1.5', className)}>
      <span
        className={cn('size-2 rounded-full shrink-0', toneDot[meta.tone])}
        aria-hidden
      />
      {withLabel && (
        <span className="font-body-sm text-[var(--color-text-secondary)]">{meta.label}</span>
      )}
      <span className="sr-only">{meta.label}</span>
    </span>
  )
}

export function Progress({

  max = 100,
  size = 'md',
  variant = 'primary',
  showLabel,
  className,

  max?: number
  size?: 'sm' | 'md' | 'lg'
  variant?: 'primary' | 'success' | 'warning' | 'error'
  showLabel?: boolean

}) {
  const pct = Math.min(100, Math.max(0, (value / max) * 100))
  const heights = { sm: 'h-1', md: 'h-1.5', lg: 'h-2.5' }
  const colors = {
    primary: 'bg-[var(--color-primary)]',

        <div className="mb-1 flex justify-between font-caption text-[var(--color-text-muted)]">

        aria-valuemin={0}
        aria-valuemax={max}

import * as React from 'react'
import { X } from 'lucide-react'
import { cn } from '@/lib/utils'
import { Button } from './button'

const sizeMap = {
  sm: 'max-w-[400px]',
  md: 'max-w-[560px]',
  lg: 'max-w-[720px]',
  xl: 'max-w-[960px]',
}

export interface ModalProps {
  open: boolean
  onClose: () => void
  title?: string
  description?: string
  size?: keyof typeof sizeMap
  children?: React.ReactNode
  footer?: React.ReactNode
  className?: string
}

export function Modal({
  open,
  onClose,
  title,
  description,
  size = 'md',
  children,
  footer,
  className,
}: ModalProps) {
  React.useEffect(() => {
    if (!open) return
    const onKey = (e: KeyboardEvent) => {
      if (e.key === 'Escape') onClose()
    }
    document.addEventListener('keydown', onKey)
    document.body.style.overflow = 'hidden'
    return () => {
      document.removeEventListener('keydown', onKey)
      document.body.style.overflow = ''
    }
  }, [open, onClose])

  if (!open) return null

  return (
    <div className="fixed inset-0 z-[var(--z-modal)] flex items-center justify-center p-4">
      <div
        className="absolute inset-0 bg-[var(--color-surface-overlay)]"
        onClick={onClose}
        aria-hidden
      />
      <div
        role="dialog"
        aria-modal="true"
        aria-labelledby={title ? 'modal-title' : undefined}
        className={cn(
          'relative z-10 w-full rounded-xl border border-[var(--color-border)] bg-[var(--color-surface)] shadow-[var(--shadow-xl)]',
          sizeMap[size],
          className,
        )}
      >
        {(title || description) && (
          <div className="flex items-start justify-between gap-3 border-b border-[var(--color-border)] px-5 py-4">
            <div>
              {title && (
                <h2 id="modal-title" className="font-h3 text-[var(--color-text-primary)]">
                  {title}
                </h2>
              )}
              {description && (
                <p className="mt-1 font-body-md text-[var(--color-text-muted)]">{description}</p>
              )}
            </div>
            <Button variant="ghost" size="icon-sm" onClick={onClose} aria-label="Close">
              <X />
            </Button>
          </div>
        )}
        {children && <div className="px-5 py-4">{children}</div>}
        {footer && (
          <div className="flex items-center justify-end gap-2 border-t border-[var(--color-border)] px-5 py-3">
            {footer}
          </div>
        )}
      </div>
    </div>
  )
}

export function ConfirmDialog({
  open,
  onClose,
  onConfirm,
  title,
  description,
  confirmLabel = 'Confirm',
  cancelLabel = 'Cancel',
  destructive,
  loading,
}: {
  open: boolean
  onClose: () => void
  onConfirm: () => void
  title: string
  description?: string
  confirmLabel?: string

import { ChevronLeft, ChevronRight, ChevronsLeft, ChevronsRight } from 'lucide-react'
import { cn } from '@/lib/utils'
import { Button } from './button'
import { Select } from './select'

export interface PaginationProps {
  page: number
  pageSize: number
  total: number
  onPageChange: (page: number) => void
  onPageSizeChange?: (size: number) => void
  pageSizeOptions?: number[]
  className?: string
}

export function Pagination({
  page,
  pageSize,
  total,
  onPageChange,
  onPageSizeChange,
  pageSizeOptions = [10, 25, 50, 100],
  className,
}: PaginationProps) {
  const totalPages = Math.max(1, Math.ceil(total / pageSize))
  const from = total === 0 ? 0 : (page - 1) * pageSize + 1
  const to = Math.min(page * pageSize, total)

  return (
    <div
      className={cn(
        'flex flex-wrap items-center justify-between gap-3 px-1 py-2',
        className,
      )}
    >
      <div className="flex items-center gap-2 font-body-sm text-[var(--color-text-muted)]">
        <span>
          {from}–{to} of {total}
        </span>
        {onPageSizeChange && (
          <div className="flex items-center gap-1.5">
            <span className="hidden sm:inline">Rows</span>
            <Select
              value={String(pageSize)}
              onChange={(e) => onPageSizeChange(Number(e.target.value))}
              className="h-8 w-[72px]"
              aria-label="Rows per page"
            >
              {pageSizeOptions.map((n) => (
                <option key={n} value={n}>
                  {n}
                </option>
              ))}
            </Select>
          </div>
        )}
      </div>
      <div className="flex items-center gap-1">
        <Button
          variant="ghost"
          size="icon-sm"
          disabled={page <= 1}
          onClick={() => onPageChange(1)}
          aria-label="First page"
        >
          <ChevronsLeft />
        </Button>
        <Button
          variant="ghost"
          size="icon-sm"
          disabled={page <= 1}
          onClick={() => onPageChange(page - 1)}
          aria-label="Previous page"
        >
          <ChevronLeft />
        </Button>
        <span className="px-2 font-body-sm tabular-nums text-[var(--color-text-secondary)]">
          {page} / {totalPages}
        </span>
        <Button
          variant="ghost"
          size="icon-sm"
          disabled={page >= totalPages}
          onClick={() => onPageChange(page + 1)}
          aria-label="Next page"
        >
          <ChevronRight />
        </Button>
        <Button
          variant="ghost"
          size="icon-sm"
          disabled={page >= totalPages}
          onClick={() => onPageChange(totalPages)}
          aria-label="Last page"
        >
          <ChevronsRight />
        </Button>
      </div>
    </div>
  )
}

            onClick={() => onChange(item.id)}
            className={cn(
              'relative inline-flex items-center gap-1.5 font-button transition-colors duration-150',
              'focus-visible:outline-none focus-visible:shadow-[var(--shadow-focus)] rounded-t-md',
              'disabled:opacity-50 disabled:pointer-events-none',
              size === 'sm' ? 'px-3 py-2 text-[13px]' : 'px-4 py-2.5',
              active
                ? 'text-[var(--color-primary)]'
                : 'text-[var(--color-text-muted)] hover:text-[var(--color-text-primary)]',
            )}
          >
            {item.label}
            {item.count != null && (
              <span
                className={cn(
                  'rounded-full px-1.5 py-0 text-[11px] font-medium tabular-nums',
                  active
                    ? 'bg-[var(--color-primary-subtle)] text-[var(--color-primary)]'
                    : 'bg-[var(--color-surface-secondary)] text-[var(--color-text-muted)]',
                )}
              >
                {item.count}
              </span>
            )}
            {active && (
              <span className="absolute bottom-0 left-0 right-0 h-0.5 bg-[var(--color-primary)] rounded-t-full" />
            )}
          </button>
        )
      })}
    </div>
  )
}
